import React, { useState } from 'react';
import { TradingProvider, useTrading } from './context/TradingContext';
import { Header } from './components/Header';
import { AssetTickerBar } from './components/AssetTickerBar';
import { TradingChart } from './components/TradingChart';
import { OrderBook } from './components/OrderBook';
import { OrderForm } from './components/OrderForm';
import { PositionsTable } from './components/PositionsTable';
import { BrokerVaultModal } from './components/BrokerVaultModal';
import { DepositModal } from './components/DepositModal';
import { MarketsOverview } from './components/MarketsOverview';
import { NotificationToast } from './components/NotificationToast';
import { AdminPortal } from './components/AdminPortal';
import { ShareModal } from './components/ShareModal';
import {
  Coins,
  ShieldCheck,
  TrendingUp,
  Zap,
  Megaphone,
} from 'lucide-react';

const TradingApp: React.FC = () => {
  const { lang, brokerConfig, setSelectedSymbol, adminSettings } = useTrading();
  const [activeTab, setActiveTab] = useState<string>('trade');
  const [isBrokerVaultOpen, setIsBrokerVaultOpen] = useState<boolean>(false);
  const [isDepositOpen, setIsDepositOpen] = useState<boolean>(false);
  const [isAdminOpen, setIsAdminOpen] = useState<boolean>(false);
  const [isShareOpen, setIsShareOpen] = useState<boolean>(false);

  const handleSelectAsset = (symbol: string) => {
    setSelectedSymbol(symbol);
    setActiveTab('trade');
  };

  return (
    <div className="min-h-screen bg-[#07090e] text-slate-100 flex flex-col font-sans selection:bg-emerald-500/30 selection:text-emerald-200">
      {/* Top Bar Header */}
      <Header
        activeTab={activeTab}
        setActiveTab={setActiveTab}
        onOpenBrokerVault={() => setIsBrokerVaultOpen(true)}
        onOpenDeposit={() => setIsDepositOpen(true)}
        onOpenAdmin={() => setIsAdminOpen(true)}
        onOpenShare={() => setIsShareOpen(true)}
      />

      {/* Admin Announcement Banner if Enabled */}
      {adminSettings.isAnnouncementActive && adminSettings.announcementText && (
        <div className="bg-emerald-500/10 border-b border-emerald-500/20 px-4 py-1.5 flex items-center justify-center gap-2 text-xs text-emerald-300">
          <Megaphone className="w-3.5 h-3.5 shrink-0 text-emerald-400" />
          <span className="font-medium text-center">{adminSettings.announcementText}</span>
        </div>
      )}

      {/* Live Market Ticker Ribbon */}
      <AssetTickerBar />

      {/* Owner Revenue Highlight Banner */}
      <div className="bg-gradient-to-r from-amber-500/10 via-slate-900 to-amber-500/10 border-b border-amber-500/20 px-4 py-2 flex items-center justify-between text-xs">
        <div className="flex items-center gap-2 max-w-2xl">
          <div className="w-5 h-5 rounded bg-amber-500/20 text-amber-300 flex items-center justify-center shrink-0">
            <Coins className="w-3.5 h-3.5" />
          </div>
          <span className="text-amber-200 font-medium">
            {lang === 'ar'
              ? `نظام الوساطة مفعل: كل صفقة تنفذ في المنصة تودع لك فورياً عمولة ${brokerConfig.feeRatePercent}% + $${brokerConfig.fixedFeeUsd} في خزينتك!`
              : `Broker Engine Active: Every trade execution deposits ${brokerConfig.feeRatePercent}% + $${brokerConfig.fixedFeeUsd} directly into your Broker Vault!`}
          </span>
        </div>

        <div className="flex items-center gap-3">
          <button
            onClick={() => setIsBrokerVaultOpen(true)}
            className="text-amber-300 hover:text-white font-bold underline font-mono text-[11px] cursor-pointer shrink-0"
          >
            {lang === 'ar' ? 'خزينة الأرباح ➔' : 'Broker Vault ➔'}
          </button>
          <span className="text-slate-600 hidden sm:inline">|</span>
          <button
            onClick={() => setIsAdminOpen(true)}
            className="text-slate-300 hover:text-amber-300 font-bold text-[11px] cursor-pointer hidden sm:inline"
          >
            {lang === 'ar' ? 'غرفة الإدارة 👑' : 'Control Room 👑'}
          </button>
        </div>
      </div>

      {/* Main Content Area */}
      <main className="flex-1 max-w-[1600px] w-full mx-auto px-2 sm:px-4 py-4">
        {activeTab === 'markets' ? (
          <MarketsOverview onSelectAsset={handleSelectAsset} />
        ) : activeTab === 'positions' ? (
          <div className="space-y-4">
            <PositionsTable />
          </div>
        ) : (
          <div className="flex flex-col gap-4">
            <div className="grid grid-cols-1 xl:grid-cols-12 gap-4">
              <div className="xl:col-span-8 flex flex-col gap-4">
                <TradingChart />
                <div className="hidden lg:block">
                  <PositionsTable />
                </div>
              </div>

              <div className="xl:col-span-4 flex flex-col gap-4">
                <OrderForm />
                <OrderBook />
              </div>
            </div>

            <div className="lg:hidden">
              <PositionsTable />
            </div>
          </div>
        )}
      </main>

      {/* Footer */}
      <footer className="border-t border-slate-900 bg-[#06080c] py-6 px-4 text-xs text-slate-500">
        <div className="max-w-[1600px] mx-auto flex flex-col sm:flex-row items-center justify-between gap-4">
          <div className="flex items-center gap-3">
            <span className="font-extrabold text-white text-sm">
              ApexTrade<span className="text-emerald-400">Pro</span>
            </span>
            <span className="text-slate-600">|</span>
            <span>منصة التداول والوساطة المتقدمة ذات السيولة العالية والعمولات المباشرة</span>
          </div>

          <div className="flex items-center gap-4 text-slate-400">
            <span className="flex items-center gap-1">
              <ShieldCheck className="w-3.5 h-3.5 text-emerald-400" />
              <span>حسابات معزولة ومحمية</span>
            </span>
            <span className="flex items-center gap-1">
              <Zap className="w-3.5 h-3.5 text-amber-400" />
              <span>تنفيذ فوري بلحظات</span>
            </span>
          </div>
        </div>
      </footer>

      {/* Modals */}
      <BrokerVaultModal
        isOpen={isBrokerVaultOpen}
        onClose={() => setIsBrokerVaultOpen(false)}
      />

      <DepositModal
        isOpen={isDepositOpen}
        onClose={() => setIsDepositOpen(false)}
      />

      {isAdminOpen && (
        <AdminPortal
          onClose={() => setIsAdminOpen(false)}
          onOpenBrokerVault={() => {
            setIsAdminOpen(false);
            setIsBrokerVaultOpen(true);
          }}
        />
      )}

      <ShareModal
        isOpen={isShareOpen}
        onClose={() => setIsShareOpen(false)}
      />

      <NotificationToast />
    </div>
  );
};

export default function App() {
  return (
    <TradingProvider>
      <TradingApp />
    </TradingProvider>
  );
}
